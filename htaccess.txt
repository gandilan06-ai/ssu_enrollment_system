# ============================================================
# .htaccess — Custom Error Pages + Security
# ============================================================

# Custom 404 page
ErrorDocument 404 /ssu_enrollment_system/404.php

# Prevent directory listing
Options -Indexes

# Block access to sensitive files
<FilesMatch "\.(sql|log|md|json)$">
    Order Allow,Deny
    Deny from all
</FilesMatch>

# Prevent access to error.log and error files
<Files "error.log">
    Order Allow,Deny
    Deny from all
</Files>
